1. Wejdź w katalog /addon_configs/a0d7b954_appdaemon/apps/ (użyj File Editor lub Samba).

2. Stwórz plik dash_saver.py i wklej poniższą zawartość:

import hassapi as hass
import os

class DashSaver(hass.Hass):
    def initialize(self):
        # Register API endpoints
        self.register_endpoint(self.save_dashboard_file, "save_dash")
        self.register_endpoint(self.read_file, "read_file")
        self.register_endpoint(self.list_directory, "list_dir")
        self.log("DashSaver bridge initialized and ready.")

    def _get_safe_path(self, path, base_dir="/config"):
        """
        Sanitize path to prevent directory traversal attacks.
        Returns safe absolute path or None if invalid.
        """
        if not path:
            return None
        # Remove leading slashes and normalize
        clean_path = os.path.normpath(path).lstrip(os.sep)
        # Prevent directory traversal
        if ".." in clean_path:
            return None
        full_path = os.path.join(base_dir, clean_path)
        # Ensure path is within base_dir
        if not os.path.abspath(full_path).startswith(os.path.abspath(base_dir)):
            return None
        return full_path

    def read_file(self, data, *args, **kwargs):
        """
        Read content of a file from AppDaemon config directory.
        POST /api/appdaemon/read_file
        Body: {"path": "dashboards/mydash.dash"}
        """
        if isinstance(data, dict) and data.get("method") == "GET" or not data:
            return {"status": "success", "message": "read_file endpoint is active"}, 200

        filepath = data.get("path")
        if not filepath:
            return {"status": "error", "message": "Missing 'path' parameter"}, 400

        safe_path = self._get_safe_path(filepath)
        if not safe_path:
            return {"status": "error", "message": "Invalid path"}, 400

        if not os.path.exists(safe_path):
            return {"status": "error", "message": f"File not found: {filepath}"}, 404

        if not os.path.isfile(safe_path):
            return {"status": "error", "message": f"Not a file: {filepath}"}, 400

        try:
            with open(safe_path, "r", encoding="utf-8") as f:
                content = f.read()
            self.log(f"Read file via API: {safe_path}")
            return {"status": "success", "content": content, "path": filepath}, 200
        except Exception as e:
            self.error(f"Error reading file: {e}")
            return {"status": "error", "message": str(e)}, 500

    def list_directory(self, data, *args, **kwargs):
        """
        List files and directories in AppDaemon config directory.
        POST /api/appdaemon/list_dir
        Body: {"path": "dashboards"} or {"path": ""} for root
        """
        if isinstance(data, dict) and data.get("method") == "GET" or not data:
            return {"status": "success", "message": "list_dir endpoint is active"}, 200

        dirpath = data.get("path", "")
        safe_path = self._get_safe_path(dirpath) if dirpath else "/config"
        
        if not safe_path:
            return {"status": "error", "message": "Invalid path"}, 400

        if not os.path.exists(safe_path):
            return {"status": "error", "message": f"Directory not found: {dirpath}"}, 404

        if not os.path.isdir(safe_path):
            return {"status": "error", "message": f"Not a directory: {dirpath}"}, 400

        try:
            items = []
            for item in os.listdir(safe_path):
                item_path = os.path.join(safe_path, item)
                items.append({
                    "name": item,
                    "type": "directory" if os.path.isdir(item_path) else "file",
                    "size": os.path.getsize(item_path) if os.path.isfile(item_path) else None
                })
            self.log(f"Listed directory via API: {safe_path}")
            return {"status": "success", "items": items, "path": dirpath}, 200
        except Exception as e:
            self.error(f"Error listing directory: {e}")
            return {"status": "error", "message": str(e)}, 500

    def save_dashboard_file(self, data, *args, **kwargs):
        """
        Save dashboard file to AppDaemon config directory.
        POST /api/appdaemon/save_dash
        Body: {"filename": "mydash.dash", "content": "..."}
        """
        if isinstance(data, dict) and data.get("method") == "GET" or not data:
            return {"status": "success", "message": "DashSaver bridge is active"}, 200

        filename = data.get("filename")
        content = data.get("content")
        
        if not filename or not content:
            return {"status": "error", "message": "Missing filename or content"}, 400

        base_dir = "/config/dashboards"
        if not os.path.exists(base_dir):
            base_dir = os.path.join(self.config_dir, "dashboards")

        if not os.path.exists(base_dir):
            try:
                os.makedirs(base_dir, exist_ok=True)
            except Exception as e:
                return {"status": "error", "message": f"Cannot create folder: {e}"}, 500

        filepath = os.path.join(base_dir, filename)

        try:
            with open(filepath, "w", encoding="utf-8") as f:
                f.write(content)
            self.log(f"Saved dashboard via API: {filepath}")
            return {"status": "success", "message": f"Saved {filename}"}, 200
        except Exception as e:
            self.error(f"Error saving dashboard: {e}")
            return {"status": "error", "message": str(e)}, 500
